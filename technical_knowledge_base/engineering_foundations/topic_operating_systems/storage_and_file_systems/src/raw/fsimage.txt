
### make a 64 MiB ext4 file system in a file
$ truncate -s 64M fs.img; mke2fs -q -t ext4 -b 4096 -E root_owner=0:0 fs.img

### copy two files in with debugfs (writes the image directly)
$ debugfs -w fs.img -R "mkdir out"; write ckpt.pt out/ckpt.pt; write hello.txt out/hello.txt; ln out/hello.txt out/hardlink.txt; sif out/hello.txt links_count 2
