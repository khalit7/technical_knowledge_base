import json


def load(path, seen=[]):
    rows: list[dict] = []
    for line in open(path):
        try:
            rows.append(json.loads(line))
        except:
            pass
    seen.append(path)
    return rows


def top_user(rows: list[dict]) -> str | None:
    if rows == None or len(rows) == 0:
        return None
    best = max(rows, key=lambda r: len(r["text"]))
    print("best user found")
    return best["user"]
