5. Async iterators and streams: how LLM tokens arrive

An async generator (async function*) yields values that arrive over time, and for await...of consumes them, exactly like Python's async def with yield and async for. Breaking out of the loop runs the generator's finally, which is where a stream closes its connection: below, the cleanup line prints after the full run, after the early break, and after Array.fromAsync collects a whole stream.

// async function* : an async generator (Python's async def with yield). Consume it with for await...of.
const sleep = ms => new Promise(r => setTimeout(r, ms));
async function* fakeModel(prompt) {
  try {
    for (const tok of ["The", " answer", " is", " 42", "."]) { await sleep(10); yield tok; }
  } finally {
    console.log("  (generator cleanup: close the HTTP connection here)");   // runs on break, return or throw too
  }
}
let text = "";
for await (const tok of fakeModel("q")) { text += tok; console.log("  token", JSON.stringify(tok)); }
console.log("full text:", JSON.stringify(text));

The same nine chunks: parsed one by one, 0 of 6 tokens; buffered, all 6

The nine chunks are the ones the client really received in the recorded run of j1_sse.mjs (a local server writing 19-byte slices 15 ms apart; byte lengths saved by the script); the two parsers run live in your browser on those bytes, and their final results are checked against the recorded run.

A streaming LLM API sends server-sent events (SSE): an HTTP response that stays open, carrying text lines data: {...}, each event ended by a blank line (the format, its parsing rules and how eight real parsers handle them are on {{Streaming and real-time|n:3c65c17b0d0d812daa9fdb38719fc82b}}). With fetch, response.body is a stream of Uint8Array byte chunks that you can for await over. The trap: the network decides where chunks end, not the server. A chunk can end in the middle of an event, and in the middle of a multi-byte UTF-8 character. Code that parses each chunk on its own works on a fast local test and fails in production behind a proxy. The fix is one streaming TextDecoder ({ stream: true } keeps half a character for the next chunk), a text buffer, and parsing only complete events. Official SDKs do this for you; the bug appears when you hand-roll a client, a proxy or a test.

  1. Chunk 1: cut mid-string. data: {"delta":"Hel starts like an event but ends inside the string, so JSON.parse fails: Unterminated string. Nothing is shown.
  2. Chunk 2: the rest is dropped. It starts with lo"}, the end of event 1, which does not start with data: and is thrown away; its second piece, data: {"delta, fails again. "Hello" is lost.
  3. Chunk 3: the same again. The comma's event is cut the same way: a third Unterminated string.
  4. Chunk 4: lost without an error. It begins mid-event and ends with dat; no piece starts with data: , so nothing is parsed and " wörld" disappears silently.
  5. Chunk 5: silent again. a: {"delta":"!"} and d: the "!" is lost the same way.

  2. Chunk 2: event 1 completes. Text up to the blank line is a whole event: "Hello" is shown, and the buffer keeps data: {"delta for the next chunk.

// How an LLM token stream arrives: server-sent events (SSE) over HTTP, read with fetch.
// A local server stands in for the model API: it sends six "data:" events, but the network
// delivers bytes in arbitrary chunks (here: 19-byte slices, 15 ms apart), cutting events and even characters in half.
import { createServer } from "node:http";
const TOKENS = ["Hello", ",", " wörld", "!", " 🚀", " done"];
const BODY = Buffer.from(TOKENS.map(t => `data: ${JSON.stringify({ delta: t })}\n\n`).join("") + "data: [DONE]\n\n");
const server = createServer(async (req, res) => {
  res.writeHead(200, { "content-type": "text/event-stream" });
  for (let i = 0; i < BODY.length; i += 19) { res.write(BODY.subarray(i, i + 19)); await new Promise(r => setTimeout(r, 15)); }
  res.end();
}).listen(0);
const url = `http://127.0.0.1:${server.address().port}/v1/stream`;

// Wrong: treat each chunk as text on its own and as whole events
const naive = { chunks: [], lens: [], events: [], errors: [] };
for await (const chunk of (await fetch(url)).body) {        // a web ReadableStream is an async iterable of Uint8Array
  const text = new TextDecoder().decode(chunk);             // a fresh decoder per chunk breaks split characters
  naive.chunks.push(text); naive.lens.push(chunk.length);
  for (const part of text.split("\n\n")) {
    if (!part.startsWith("data: ") || part === "data: [DONE]") continue;
    try { naive.events.push(JSON.parse(part.slice(6)).delta); } catch (e) { naive.errors.push(e.message.split(" in JSON")[0]); }
  }
}
// Right: one streaming decoder, a buffer, and only complete events (ending in a blank line) are parsed
async function* sseEvents(body) {
  const dec = new TextDecoder(); let buf = "";
  for await (const chunk of body) {
    buf += dec.decode(chunk, { stream: true });              // stream: true keeps half a character for the next chunk
    let end;
    while ((end = buf.indexOf("\n\n")) >= 0) { const ev = buf.slice(0, end); buf = buf.slice(end + 2); yield ev; }
  }
}
const good = [];
for await (const ev of sseEvents((await fetch(url)).body)) {
  const data = ev.slice(6);
  if (data === "[DONE]") break;
  good.push(JSON.parse(data).delta);
}
server.close();
console.log(`${BODY.length} bytes arrived in ${naive.chunks.length} chunks; the first three:`);
naive.chunks.slice(0, 3).forEach((c, i) => console.log(`  chunk ${i + 1}: ${JSON.stringify(c)}`));
console.log("naive parser, tokens:", JSON.stringify(naive.events.join("")), "| errors:", naive.errors.length, naive.errors.slice(0, 2));
console.log("buffered parser, tokens:", JSON.stringify(good.join("")));
if (process.argv[2]) (await import("node:fs")).writeFileSync(process.argv[2], JSON.stringify({ body: BODY.toString(), slice: 19, chunks: naive.chunks, lens: naive.lens, naive: naive.events, naiveErrors: naive.errors, good }));

Node streams and backpressure. Node also has its own older stream classes (Readable, Writable, Transform, pipeline) used by files, sockets and HTTP. They handle backpressure: a fast producer is paused while a slow consumer catches up, so memory stays bounded. Each stream has a buffer with a highWaterMark; push() and write() return false once it is reached, and pipeline() obeys that signal for you. Readable streams are async iterable too, which is the easiest way to use them. Fig. 3 replays a run of k1_backpressure.mjs recorded for this page (Node 20.20.1 on an AMD Ryzen 9 9950X under Linux, 2026-11-07; every event and buffer level was the same in four runs, the totals within 2 ms). What happens to a server's memory when it ignores write()'s false at scale, measured, is on {{TypeScript: LLM apps and agents|n:4055c17b0d0d8109aad6e69a87b3bd6f}}.

  1. Three stages, four buffers. A source (Readable), a pass-through Transform and a sink (Writable) that takes 20 ms for each 1 KiB chunk, joined by pipeline(). Every buffer has a highWaterMark of 2 KiB; the Transform has one on each side. At 1.1 ms the source's read() is called for the first time.
  2. The source fills its buffer and stops. push() returns true for chunk 0 and false for chunk 1: the buffer has reached its 2 KiB mark, so the source stops pushing. The mark is a threshold, not a cap: the chunk that crossed it is kept.
  3. Chunks flow down until every buffer is at its mark. The pipe moves chunks into the Transform and the sink (the sink starts writing chunk 0 at 1.4 ms), and each time the source's buffer drops below its mark, read() asks for one more. By 1.7 ms every buffer is full; the four counters add up to 8 KiB.
  4. Paused for 20 ms. From 1.7 to 22.0 ms nobody calls read(): the source is paused, and the only work is the sink writing chunk 0.
  5. Room travels up the pipe. The sink finishes chunks 0 and 1 (22.0 and 42.2 ms); the space they leave moves up stage by stage, and at 42.6 ms read() is called twice, for chunks 7 and 8.
  6. Steady state, paced by the sink. From then on, about every 40 ms, the sink finishes two chunks and the source pushes two more, up to chunk 19 at 283.9 ms. No buffer ever holds more than its mark plus the one chunk that crossed it.
  7. The end. The source pushes null (end of stream), the buffers drain, and the sink finishes chunk 19 at 403.9 ms.
  1. The same sink, written to in a loop. A for loop calls sink.write() for each of the 20 chunks and ignores what it returns. The first write returns true, and the sink starts on chunk 0 at once.
  2. write() says false 19 times; the loop goes on. From the second chunk on, write() returns false: the buffer is at its 2 KiB mark. By 0.1 ms all 20 chunks sit in the sink's buffer: 20 KiB, ten times the mark. Memory now grows with the input, not with the mark.
  3. Drained one chunk every 20 ms. The sink still takes 20 ms a chunk; at 201.4 ms it has finished 10 and 10 KiB are still waiting.
  4. The same time, 2.5 times the memory. The last chunk finishes at 402.2 ms, against 403.9 ms with pipeline(): the sink sets the pace either way, so ignoring false bought no speed, only memory (20 KiB against 8 KiB here; for a long stream to a slow client, without bound).

k1_backpressure.mjs
// Backpressure in a Node stream pipeline: a fast source, a pass-through Transform, a slow sink (20 ms per 1 KiB chunk).
// Every buffer's highWaterMark is 2 KiB. Mode "pipeline": stream.pipeline() pauses the source when the buffers are full.
// Mode "ignore": a loop calls sink.write() for every chunk and ignores the false it returns, so the sink's buffer grows.
// Each event is stamped with ms since the start and the bytes waiting in every buffer; the Reading stage replays them.
import { Readable, Transform, Writable } from "node:stream";
import { pipeline } from "node:stream/promises";
import { writeFileSync } from "node:fs";
const N = 20, KB = 1024, HWM = 2 * KB, SLOW = 20;
const chunk = i => Buffer.alloc(KB, 97 + (i % 26));
async function run(mode) {
  const EV = [], T0 = performance.now(); let src = null, xf = null, sink = null;
  const lv = () => [src ? src.readableLength : 0, xf ? xf.writableLength : 0, xf ? xf.readableLength : 0, sink.writableLength];
  const ev = (what, i, ok) => EV.push([what, i, +(performance.now() - T0).toFixed(1), ...lv(), ok === undefined ? null : ok]);
  sink = new Writable({ highWaterMark: HWM, write(c, _e, cb) { const i = c[0] - 97; ev("write", i); setTimeout(() => { ev("done", i); cb(); }, SLOW); } });
  if (mode === "pipeline") {
    let i = 0;
    src = new Readable({ highWaterMark: HWM, read() {             // called whenever the source's buffer has room
      ev("read", i);
      while (i < N) { const ok = this.push(chunk(i)); ev("push", i, ok); i++; if (!ok) return; }   // false: stop until read() is called again
      this.push(null);
    } });
    xf = new Transform({ highWaterMark: HWM, transform(c, _e, cb) { cb(null, c); } });
    await pipeline(src, xf, sink);
  } else {
    let falses = 0;
    for (let i = 0; i < N; i++) { const ok = sink.write(chunk(i)); ev("push", i, ok); if (!ok) falses++; }   // ignores false
    await new Promise(r => sink.end(r));
    ev("end", N);
    console.log(`ignore: write() returned false ${falses} times`);
  }
  const peak = Math.max(...EV.map(e => e[3] + e[4] + e[5] + e[6]));
  console.log(`${mode}: ${N} chunks of 1 KiB, highWaterMark 2 KiB, sink 20 ms a chunk: ${EV.at(-1)[2].toFixed(0)} ms, peak buffered ${(peak / KB).toFixed(0)} KiB, peak in the sink ${(Math.max(...EV.map(e => e[6])) / KB).toFixed(0)} KiB`);
  return { events: EV, total_ms: EV.at(-1)[2], peak_kib: peak / KB };
}
const out = { n: N, hwm_kib: HWM / KB, slow_ms: SLOW, cols: ["what", "chunk", "ms", "source", "transform in", "transform out", "sink"] };
out.pipeline = await run("pipeline");
out.ignore = await run("ignore");
if (process.argv[2]) writeFileSync(process.argv[2], JSON.stringify(out));
